import { z } from 'zod';
import { isoDate } from '../../common/utils/validators.js';

const colour = z.string().regex(/^#[0-9a-f]{6}$/i);
const mediaUrl = z.string().max(1000).regex(/^(https:\/\/|\/(?!\/))/).or(z.literal(''));

/** Settings for Sunset Journey only. Other invitation schemas stay independent. */
export const sunsetJourneySettingsSchema = z.object({
  main: colour,
  gold: colour,
  paper: colour,
  sky: colour,
  font: z.enum(['classic', 'script', 'modern']),
  story: z.string().max(2000),
  familiesLine: z.string().max(2000),
  timeline: z.array(z.object({
    year: z.string().max(20),
    title: z.string().max(150),
    description: z.string().max(1000),
  })).max(10),
  eventDates: z.array(isoDate.or(z.literal(''))).max(20),
  musicUrl: mediaUrl,
  filmVideoUrl: mediaUrl,
}).partial();
