import { describe, expect, it } from 'vitest';
import { updateInvitationSchema } from '../src/modules/invitations/invitation.validation.js';
import { rsvpSchema } from '../src/modules/guests/guest.validation.js';

describe('Gilded Celebration', () => {
  it.each(['classic', 'script', 'modern'])('saves %s without changing Premium settings', font => {
    const extraDetails = {
      gilded: { wine: '#781c32', gold: '#b88a43', ivory: '#faf7f0', ink: '#42332f', muted: '#84756c', font,
        storyHeading: 'Our story', storyText: 'Together', quote: 'Forever', musicUrl: '',
        eventDates: ['2027-02-20', '', '2027-02-22'], eventMaps: ['https://maps.google.com/?q=Colombo', ''] },
      sunset: { font: 'classic' }, floralEnvelope: { palette: 'burgundy' }, heirloom: { palette: 'heritage-gold' },
    };
    expect(updateInvitationSchema.parse({ extraDetails }).extraDetails).toEqual(extraDetails);
  });
  it.each([{ wine: 'red' }, { font: 'unknown' }, { eventDates: ['2027-02-30'] },
    { eventMaps: ['javascript:alert(1)'] }, { musicUrl: '//example.com/song.mp3' },
    { storyText: 'a'.repeat(4001) }])('rejects invalid design settings: %j', invalid => {
    expect(updateInvitationSchema.safeParse({ extraDetails: { gilded: invalid } }).success).toBe(false);
  });
  it('keeps an optional guest email and normalizes it', () => {
    expect(rsvpSchema.parse({ name: 'Guest', email: 'GUEST@EXAMPLE.COM', attending: true, headCount: 2 }).email).toBe('guest@example.com');
  });
  it('still accepts the existing RSVP format', () => {
    expect(rsvpSchema.parse({ name: 'Guest', attending: false, headCount: 1 }).email).toBeUndefined();
  });
  it('rejects an invalid RSVP email', () => {
    expect(rsvpSchema.safeParse({ name: 'Guest', email: 'invalid', attending: true, headCount: 1 }).success).toBe(false);
  });
});
